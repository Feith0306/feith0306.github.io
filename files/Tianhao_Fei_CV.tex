%-----------RESEARCH Experience-----------
\section{\textsc{\Large{Research Experience}}}
\begin{enumerate}[leftmargin=0.2in]
	\small{
		\item {\textbf{\externallink{https://feith0306.github.io/about/research/backstepping-control/}{Adaptive Backstepping Control for Constrained Uncertain Nonlinear Systems}}}
		\begin{itemize}[leftmargin=0in]
			\item % \textbf{Stochastic nonlinear control:}
			Developed adaptive neural backstepping controllers for stochastic nonlinear systems with uncertain dynamics and time-varying constraints, ensuring stability and constraint satisfaction.
			\item %\textbf{Distributed synchronization of multi-agent systems:}
			Designed command-filtered adaptive controllers with compensation mechanisms for heterogeneous switched multi-agent systems under asynchronous switching, time-varying delays.
			\item %\textbf{Game-theoretic robust tracking:}
			Formulated robust $H_\infty$ tracking as a zero-sum differential game and developed Actor-Critic-Disturbance learning schemes combined with backstepping control to address nonlinear dynamics and disturbances.
			\item \textbf{Outcome:}
			\textbf{3 journal articles}
			(\textit{Neurocomputing},
			\textit{J. Franklin Inst.},
			\textit{Int. J. Robust Nonlinear Control}).
		\end{itemize}
		
		\item {\textbf{\externallink{https://feith0306.github.io/about/research/projected-self-triggered-learning/}{Projected Self-Triggered Reinforcement Learning for Differential Games}}}
		\begin{itemize}[leftmargin=0in]
			\item %\textbf{Self-triggered learning:} 
			Proposed communication-efficient learning frameworks for zero-sum and nonzero-sum differential games, avoiding continuous state monitoring while preserving closed-loop stability.
			\item %\textbf{Projected safe adaptation:} 
			Incorporated projection operators into Actor–Critic and Actor–Critic–Disturbance learning laws to prevent parameter drift and keep adaptive weights within prescribed compact sets.
			\item %\textbf{Resource-aware Nash approximation:} 
			Combined intermittent feedback, fuzzy logic, and experience replay to approximate Nash policies with reduced sampling, bounded learning errors, and Zeno-free execution.
			\item \textbf{Outcome:} \textbf{2 journal articles} (\textit{IEEE Trans. Fuzzy Syst.}, \textit{IEEE Trans. Emerg. Topics Comput. Intell.}); \textbf{3 conference papers}.
		\end{itemize}
		\item {\textbf{\externallink{https://feith0306.github.io/about/research/composite-actor-critic-learning/}{Data-Efficient Composite Actor–Critic Reinforcement Learning}}}
		\begin{itemize}[leftmargin=0in]
			\item %\textbf{Composite learning for game-theoretic problem:}
			Developed Actor–Critic learning algorithm for multi-player nonzero-sum differential games by integrating instantaneous Bellman errors with historical data.
			\item %\textbf{Relaxed excitation requirements:} 
			Replaced restrictive persistence of excitation condition with finite/sufficient excitation-based learning, improving the practicality of online adaptive dynamic programming.
			\item %\textbf{Improved convergence and stability:} 
			Established convergence of critic/actor weights and closed-loop stability while enabling approximate Nash equilibrium learning for nonlinear multi-agent decision systems.
			\item \textbf{Outcome:} \textit{Submitted to IEEE Trans. Neural Netw. Learn. Syst}.
		\end{itemize}
		}
\end{enumerate}
